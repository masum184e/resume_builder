% One-column entries
\newenvironment{onecolentry}
    {\begin{adjustwidth}{0cm + 0.00001cm}{0cm + 0.00001cm}}
    {\end{adjustwidth}}

% Two-column entries
\newenvironment{twocolentry}[2][]
    {%
        \onecolentry
        \def\secondColumn{#2}%
        \setcolumnwidth{\fill, 4.5cm}%
        \begin{paracol}{2}%
    }
    {%
        \switchcolumn
        \raggedleft
        \secondColumn
        \end{paracol}%
        \endonecolentry
    }

% Three-column entries
\newenvironment{threecolentry}[3][]
    {%
        \onecolentry
        \def\thirdColumn{#3}%
        \setcolumnwidth{\fill, \fill, 4.5cm}%
        \begin{paracol}{3}%
        {\raggedright #2}%
        \switchcolumn
    }
    {%
        \switchcolumn
        \raggedleft
        \thirdColumn
        \end{paracol}%
        \endonecolentry
    }
